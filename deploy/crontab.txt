# Crons RushPlay (Coolify « Scheduled Tasks », service api). Heures en UTC.
# relevé de cotes : 08:00 UTC, 4 jours sur 7 (dimanche, lundi, mercredi, vendredi) — 8 compétitions à cotes live
# (top 5, LdC, Ligue Europa, Ligue des Nations) x 3 crédits (2 h2h + 1 totals over/under) = 24 crédits par relevé,
# ≈ 17 relevés/mois ≈ 408 crédits/mois, sous le plafond du plan gratuit (500/mois). À 5 jours sur 7 on dépasserait
# (≈ 528). Les 11 championnats secondaires sont en mode gratuit (fd_uk seul) : ils ne coûtent aucun crédit.
0 8 * * 0,1,3,5   cd /app && python -m app.collectors.run odds
# calendrier, heures, résultats, reports, règlement des paris : tous les jours 06:30 UTC (football-data.org, gratuit)
30 6 * * *  cd /app && python -m app.collectors.run fd_org
# matchs de la semaine avec cotes moyennes (repli quand le relevé Odds API n'a pas eu lieu) : tous les jours 06:45 UTC
45 6 * * *  cd /app && python -m app.collectors.run fixtures
# historique, résultats et cotes ouverture/clôture des 16 championnats fd_uk : lundi 07:00 UTC. Sans argument,
# la saison en cours est calculée (août → juillet) : ne plus coder « 2526 » en dur, c'était la saison précédente.
# C'est la SEULE source de résultats pour les championnats sans calendrier football-data.org (D2, I2, SP2, F2, B1,
# T1, G1, SC0) : sans ce cron, leurs matchs restent « programmés » et les paris du carnet ne se règlent pas.
0 7 * * 1   cd /app && python -m app.collectors.run fd_uk
# fusion des doublons entre sources : tous les jours 07:15 UTC, après les imports
15 7 * * *  cd /app && python -m app.collectors.run dedup
